# Development/CI-only dependencies. Kept separate from requirements.txt so an
# end-user install (./install.sh) never pulls the test toolchain.
#
#   pip install -r scripts/dev/requirements-dev.txt
#
# pytest >= 7 is required for the `pythonpath` ini option used in pyproject.toml.
pytest>=7.0
# Coverage visibility (report-only; CI prints the summary, no threshold gate).
pytest-cov
# Shell linting: provides a `shellcheck` binary in the venv so
# scripts/dev/check.sh can run the canonical shell gate locally.
shellcheck-py
# Static contract checking (reads [tool.pyright] from pyproject.toml).
basedpyright
# Fast Python linting and formatting (configured for the Python 3.10 floor).
ruff
